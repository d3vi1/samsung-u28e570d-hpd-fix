cmake_minimum_required(VERSION 3.15)
project(SamsungMonitorFirmware VERSION 0.1.0 LANGUAGES CXX)

option(SAMSUNG_TARGET_XP "Target Windows XP x86 with a compatible toolchain" OFF)
option(SAMSUNG_BUILD_GUI "Build the native Win32 application" ON)
option(SAMSUNG_ENABLE_HARDWARE "Enable explicit NVAPI access in the application (not in tests)" OFF)
option(SAMSUNG_BUILD_TESTS "Build offline tests" ON)

set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(SAMSUNG_TARGET_XP)
    if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
        message(FATAL_ERROR "The XP profile requires a Windows x86 toolchain.")
    endif()
    if(MSVC AND NOT CMAKE_VS_PLATFORM_TOOLSET MATCHES "_xp$")
        message(FATAL_ERROR "MSVC XP builds require an XP toolset such as v141_xp.")
    endif()
    add_compile_definitions(WINVER=0x0501 _WIN32_WINNT=0x0501)
else()
    add_compile_definitions(WINVER=0x0601 _WIN32_WINNT=0x0601)
endif()
if(WIN32)
    add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
if(MSVC)
    add_compile_options(/W4 /EHsc)
else()
    add_compile_options(-Wall -Wextra -Wpedantic)
endif()

add_library(monitorfw_core STATIC
    src/core.cpp src/package.cpp src/sha256.cpp
    src/nvapi_transport.cpp src/mstar_flash.cpp src/provider_registry.cpp src/broker_protocol.cpp)
target_include_directories(monitorfw_core PUBLIC include)

if(WIN32 AND SAMSUNG_BUILD_GUI)
    add_library(monitorfw_windows_store STATIC src/win32_store.cpp)
    target_include_directories(monitorfw_windows_store PUBLIC include)
    target_link_libraries(monitorfw_windows_store PUBLIC monitorfw_core shell32 ole32)
endif()

if(SAMSUNG_BUILD_GUI AND WIN32)
    enable_language(RC)
    if(MINGW)
        # windres forwards include paths through its own preprocessor command.
        # Run in the resource directory with relative inputs so paths containing
        # spaces are not reparsed by that command. The build shell has GCC on PATH.
        set(APP_RESOURCES "${CMAKE_CURRENT_BINARY_DIR}/app-resources.obj")
        add_custom_command(OUTPUT "${APP_RESOURCES}"
            COMMAND "${CMAKE_RC_COMPILER}" --preprocessor=gcc
                --preprocessor-arg=-E --preprocessor-arg=-xc --preprocessor-arg=-DRC_INVOKED
                -I . -O coff -i app.rc -o "${APP_RESOURCES}"
            WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/resources"
            DEPENDS resources/app.rc resources/app.manifest VERBATIM)
        set_source_files_properties("${APP_RESOURCES}" PROPERTIES GENERATED TRUE EXTERNAL_OBJECT TRUE)
    else()
        set(APP_RESOURCES resources/app.rc)
    endif()
    add_executable(samsung_monitor_firmware WIN32 src/win32_main.cpp "${APP_RESOURCES}")
    target_include_directories(samsung_monitor_firmware PRIVATE tests resources)
    target_compile_definitions(samsung_monitor_firmware PRIVATE
        MONITORFW_ENABLE_HARDWARE=$<BOOL:${SAMSUNG_ENABLE_HARDWARE}>)
    target_link_libraries(samsung_monitor_firmware PRIVATE
        monitorfw_windows_store comctl32 comdlg32 shell32 ole32 gdi32 user32)
    if(MINGW)
        target_link_options(samsung_monitor_firmware PRIVATE -municode -static -static-libgcc -static-libstdc++)
        if(SAMSUNG_TARGET_XP)
            target_link_options(samsung_monitor_firmware PRIVATE -Wl,--major-subsystem-version,5,--minor-subsystem-version,1)
        endif()
    elseif(MSVC AND SAMSUNG_TARGET_XP)
        target_link_options(samsung_monitor_firmware PRIVATE /SUBSYSTEM:WINDOWS,5.01)
    endif()
    add_executable(samsung_monitor_broker WIN32 src/desktop_broker.cpp "${APP_RESOURCES}")
    target_include_directories(samsung_monitor_broker PRIVATE tests resources)
    target_compile_definitions(samsung_monitor_broker PRIVATE
        MONITORFW_ENABLE_HARDWARE=$<BOOL:${SAMSUNG_ENABLE_HARDWARE}>)
    target_link_libraries(samsung_monitor_broker PRIVATE
        monitorfw_windows_store shell32 ole32 advapi32 user32)
    if(MINGW)
        target_link_options(samsung_monitor_broker PRIVATE -municode -static -static-libgcc -static-libstdc++)
        if(SAMSUNG_TARGET_XP)
            target_link_options(samsung_monitor_broker PRIVATE -Wl,--major-subsystem-version,5,--minor-subsystem-version,1)
        endif()
    elseif(MSVC AND SAMSUNG_TARGET_XP)
        target_link_options(samsung_monitor_broker PRIVATE /SUBSYSTEM:WINDOWS,5.01)
    endif()
endif()

if(SAMSUNG_BUILD_TESTS)
    enable_testing()
    foreach(test_name IN ITEMS core_tests transport_tests broker_tests package_conformance)
        if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/${test_name}.cpp")
            add_executable(${test_name} tests/${test_name}.cpp)
            target_link_libraries(${test_name} PRIVATE monitorfw_core)
            target_include_directories(${test_name} PRIVATE tests)
            if(MINGW)
                target_link_options(${test_name} PRIVATE -static -static-libgcc -static-libstdc++)
                if(SAMSUNG_TARGET_XP)
                    target_link_options(${test_name} PRIVATE -Wl,--major-subsystem-version,5,--minor-subsystem-version,1)
                endif()
            elseif(MSVC AND SAMSUNG_TARGET_XP)
                target_link_options(${test_name} PRIVATE /SUBSYSTEM:CONSOLE,5.01)
            endif()
            add_test(NAME ${test_name} COMMAND ${test_name})
        endif()
    endforeach()
    if(WIN32 AND SAMSUNG_BUILD_GUI)
        add_executable(storage_tests tests/storage_tests.cpp)
        target_include_directories(storage_tests PRIVATE tests)
        target_link_libraries(storage_tests PRIVATE monitorfw_windows_store)
        if(MINGW)
            target_link_options(storage_tests PRIVATE -static -static-libgcc -static-libstdc++)
            if(SAMSUNG_TARGET_XP)
                target_link_options(storage_tests PRIVATE -Wl,--major-subsystem-version,5,--minor-subsystem-version,1)
            endif()
        elseif(MSVC AND SAMSUNG_TARGET_XP)
            target_link_options(storage_tests PRIVATE /SUBSYSTEM:CONSOLE,5.01)
        endif()
        add_test(NAME storage_tests COMMAND storage_tests)
        add_test(NAME ui_mock_smoke COMMAND samsung_monitor_firmware --smoke-test)
        set_tests_properties(ui_mock_smoke PROPERTIES TIMEOUT 60)
        find_program(TEST_POWERSHELL NAMES pwsh.exe powershell.exe)
        if(NOT TEST_POWERSHELL)
            message(FATAL_ERROR "Broker smoke tests require PowerShell on the build/test host.")
        endif()
        add_test(NAME broker_mock_smoke COMMAND "${TEST_POWERSHELL}" -NoProfile -File
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/Run-BrokerSmoke.ps1" -BrokerExe "$<TARGET_FILE:samsung_monitor_broker>")
        set_tests_properties(broker_mock_smoke PROPERTIES TIMEOUT 60)
    endif()
endif()
